# First-time server setup (already done on srv1519994; kept for reference / a new server).
bash <<'EOF'
set -e
[ "$(id -u)" = 0 ] || { echo "Please run as root (type: sudo -i), then paste again."; exit 1; }
apt-get update -qq
apt-get install -y git curl
if ! command -v node >/dev/null || [ "$(node -p 'process.versions.node.split(".")[0]')" -lt 18 ]; then
  curl -fsSL https://deb.nodesource.com/setup_20.x | bash -
  apt-get install -y nodejs
fi
command -v npm >/dev/null || apt-get install -y npm
id potatobrawl >/dev/null 2>&1 || adduser --system --group --no-create-home potatobrawl
git config --global --add safe.directory /opt/potatobrawl
if [ -d /opt/potatobrawl/.git ]; then
  git -C /opt/potatobrawl pull --ff-only
else
  [ -e /opt/potatobrawl ] && mv /opt/potatobrawl /opt/potatobrawl.old.$(date +%s)
  git clone https://github.com/guspliffy/potatobrawl /opt/potatobrawl
fi
cd /opt/potatobrawl
mkdir -p public
for f in index.html three.min.js; do [ -f "$f" ] && cp -f "$f" public/; done
[ -f public/index.html ] && [ -f public/three.min.js ] || { echo "index.html or three.min.js is missing on GitHub."; exit 1; }
npm install --omit=dev
chown -R potatobrawl:potatobrawl /opt/potatobrawl
cp potatobrawl.service /etc/systemd/system/
systemctl daemon-reload
systemctl enable potatobrawl
systemctl restart potatobrawl
ufw allow 8080/tcp >/dev/null 2>&1 || true
sleep 2
curl -s localhost:8080/health
EOF
